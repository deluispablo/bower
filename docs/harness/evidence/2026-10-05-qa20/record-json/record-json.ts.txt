/** Strict record JSON and canonical identities; no schema or permission authority. */
import { sha256 } from '../vault/hashes';

export type RecordJsonValue = null | boolean | number | string | RecordJsonValue[] | { [key: string]: RecordJsonValue };
export type RecordJsonFailure = { ok: false; reason: 'encoding' | 'syntax' | 'duplicate-key' | 'unicode' | 'number' | 'value' };
export type RecordJsonResult<T> = { ok: true; value: T } | RecordJsonFailure;

class Refusal extends Error {
  constructor(readonly reason: RecordJsonFailure['reason']) { super(`Record JSON refused: ${reason}`); }
}

function validString(text: string): void {
  for (let i = 0; i < text.length; i++) {
    const code = text.charCodeAt(i);
    if (code >= 0xd800 && code <= 0xdbff) {
      const next = text.charCodeAt(++i);
      if (!(next >= 0xdc00 && next <= 0xdfff)) throw new Refusal('unicode');
    } else if (code >= 0xdc00 && code <= 0xdfff) throw new Refusal('unicode');
  }
}

function validNumber(value: number): void {
  if (!Number.isSafeInteger(value) || value < 0 || Object.is(value, -0)) throw new Refusal('number');
}

/** Recognized validation failures return no value; unexpected/platform errors propagate. */
function refused(error: unknown): RecordJsonFailure {
  if (error instanceof Refusal) return { ok: false, reason: error.reason };
  throw error;
}

class Reader {
  private at = 0;
  constructor(private readonly text: string) {}

  read(): RecordJsonValue {
    const value = this.value();
    this.space();
    if (this.at !== this.text.length) throw new Refusal('syntax');
    return value;
  }

  private space(): void {
    while ([' ', '\t', '\n', '\r'].includes(this.text.charAt(this.at))) this.at++;
  }

  private take(char: string): void {
    this.space();
    if (this.text.charAt(this.at++) !== char) throw new Refusal('syntax');
  }

  private string(): string {
    const start = this.at;
    this.take('"');
    while (this.at < this.text.length) {
      const char = this.text.charAt(this.at++);
      if (char === '\\') this.at++;
      else if (char === '"') {
        let value: unknown;
        try { value = JSON.parse(this.text.slice(start, this.at)); }
        catch (error) {
          if (error instanceof SyntaxError) throw new Refusal('syntax');
          throw error;
        }
        if (typeof value !== 'string') throw new Refusal('syntax');
        validString(value);
        return value;
      }
    }
    throw new Refusal('syntax');
  }

  private number(): number {
    const match = /^-?(?:0|[1-9]\d*)(?:\.(\d+))?(?:[eE]([+-]?\d+))?/.exec(this.text.slice(this.at));
    if (!match) throw new Refusal('syntax');
    const token = match[0];
    const value = Number(token);
    validNumber(value);
    // Number() can round a fractional decimal to a safe integer. Verify its
    // exact decimal domain before allowing that lossy conversion to authorize it.
    const coefficient = token.split(/[eE]/)[0] ?? '';
    const digits = coefficient.replace(/[-.]/g, '').replace(/^0+/, '');
    if (digits) {
      const scale = Number(match[2] ?? '0') - (match[1]?.length ?? 0);
      if (!Number.isFinite(scale)) throw new Refusal('number');
      if (scale < 0) {
        const cut = digits.length + scale;
        if (cut <= 0 || /[1-9]/.test(digits.slice(cut))) throw new Refusal('number');
      }
    }
    this.at += token.length;
    return value;
  }

  private value(): RecordJsonValue {
    this.space();
    const char = this.text.charAt(this.at);
    if (char === '"') return this.string();
    if (char === '{') {
      this.at++;
      const result = Object.create(null) as { [key: string]: RecordJsonValue };
      const keys = new Set<string>();
      this.space();
      if (this.text.charAt(this.at) === '}') { this.at++; return result; }
      for (;;) {
        this.space();
        if (this.text.charAt(this.at) !== '"') throw new Refusal('syntax');
        const key = this.string();
        if (keys.has(key)) throw new Refusal('duplicate-key');
        keys.add(key);
        this.take(':');
        result[key] = this.value();
        this.space();
        if (this.text.charAt(this.at) === '}') { this.at++; return result; }
        this.take(',');
      }
    }
    if (char === '[') {
      this.at++;
      const result: RecordJsonValue[] = [];
      this.space();
      if (this.text.charAt(this.at) === ']') { this.at++; return result; }
      for (;;) {
        result.push(this.value());
        this.space();
        if (this.text.charAt(this.at) === ']') { this.at++; return result; }
        this.take(',');
      }
    }
    for (const [token, value] of [['true', true], ['false', false], ['null', null]] as const) {
      if (this.text.startsWith(token, this.at)) { this.at += token.length; return value; }
    }
    return this.number();
  }
}

/** Invalid UTF-8 and decoded duplicate keys are refused before producing data. */
export function parseRecordJson(bytes: Uint8Array): RecordJsonResult<RecordJsonValue> {
  let text: string;
  try { text = new TextDecoder('utf-8', { fatal: true, ignoreBOM: true }).decode(bytes); }
  catch (error) {
    if (error instanceof TypeError) return { ok: false, reason: 'encoding' };
    throw error;
  }
  try { return { ok: true, value: new Reader(text).read() }; }
  catch (error) {
    // expected: recognized validation refuses explicitly without data; refused rethrows unexpected errors.
    return refused(error);
  }
}

function emit(value: unknown, ancestors: Set<object>): string {
  if (value === null) return 'null';
  if (typeof value === 'boolean') return value ? 'true' : 'false';
  if (typeof value === 'string') { validString(value); return JSON.stringify(value); }
  if (typeof value === 'number') { validNumber(value); return String(value); }
  if (typeof value !== 'object') throw new Refusal('value');
  if (ancestors.has(value)) throw new Refusal('value');
  ancestors.add(value);
  try {
    const descriptors = Object.getOwnPropertyDescriptors(value);
    const keys = Reflect.ownKeys(descriptors);
    if (keys.some((key) => typeof key !== 'string')) throw new Refusal('value');
    const property = (key: string): unknown => {
      const descriptor = descriptors[key];
      if (!descriptor || !('value' in descriptor) || !descriptor.enumerable) throw new Refusal('value');
      return descriptor.value as unknown;
    };
    if (Array.isArray(value)) {
      if (keys.length !== value.length + 1 || !Object.hasOwn(descriptors, 'length')) throw new Refusal('value');
      const parts: string[] = [];
      for (let i = 0; i < value.length; i++) parts.push(emit(property(String(i)), ancestors));
      return `[${parts.join(',')}]`;
    }
    if (Object.getPrototypeOf(value) !== Object.prototype && Object.getPrototypeOf(value) !== null) throw new Refusal('value');
    // Direct emission is essential: JSON.stringify(object) reorders integer keys.
    const parts: string[] = [];
    for (const key of Object.getOwnPropertyNames(descriptors).sort()) {
      validString(key);
      parts.push(`${JSON.stringify(key)}:${emit(property(key), ancestors)}`);
    }
    return `{${parts.join(',')}}`;
  } finally { ancestors.delete(value); }
}

/** Canonical UTF-8 bytes; every returned buffer is independent of earlier calls. */
export function canonicalRecordJson(value: unknown): RecordJsonResult<Uint8Array> {
  try { return { ok: true, value: new TextEncoder().encode(emit(value, new Set())) }; }
  catch (error) {
    // expected: recognized validation refuses explicitly without data; refused rethrows unexpected errors.
    return refused(error);
  }
}

/** Hash canonical JSON bytes, never the raw content bytes of a string. */
export async function hashRecordJson(value: unknown): Promise<RecordJsonResult<string>> {
  const canonical = canonicalRecordJson(value);
  if (!canonical.ok) return canonical;
  return { ok: true, value: await sha256(new Blob([new Uint8Array(canonical.value).buffer])) };
}

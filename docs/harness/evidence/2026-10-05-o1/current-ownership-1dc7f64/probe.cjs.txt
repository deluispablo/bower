const fs = require('node:fs');
const assert = require('node:assert/strict');
const root = process.argv[2]; const ts = require(process.argv[3]);
require.extensions['.ts'] = (m,p) => m._compile(ts.transpileModule(fs.readFileSync(p,'utf8'),{compilerOptions:{module:ts.ModuleKind.CommonJS,target:ts.ScriptTarget.ES2022}}).outputText,p);
const {parseNote,serializeNote,toggleTaskInText}=require(root+'/src/data/vault/note-format.ts');
const canonical=serializeNote({summary:'Synthetic example',properties:[],tags:[],byBower:true,currentState:['Summary from Bower.'],body:'A personal paragraph.\n- [ ] Follow up personally',timeline:[{date:'5 Oct',text:'Synthetic history.'}]});
const c=parseNote('Example.md',canonical); const tick=toggleTaskInText(canonical,0);
assert.deepEqual(c.currentState,['Summary from Bower.'],'control: current producer bullet grammar');
assert.equal(c.body,'A personal paragraph.\n- [ ] Follow up personally','control: canonical personal body retained');
assert.equal(tick.label,'Follow up personally','control: canonical personal task editable');
assert.equal(tick.text,canonical.replace('- [ ] Follow up personally','- [x] Follow up personally'),'control: only intended checkbox byte changes');
const crlf=canonical.replace(/\n/g,'\r\n');assert.equal(toggleTaskInText(crlf,0).text,crlf.replace('- [ ] Follow up personally','- [x] Follow up personally'),'control: closed CRLF byte preservation');
const historical='---\nsummary: Synthetic example\nby: bower\n---\n\n## Current state\nA paragraph maintained by Bower.\n\n- [ ] protected task\n\n## My notes\n- [ ] my task\n\n## Timeline\n- 5 Oct · Synthetic history.';
const h=parseNote('Example.md',historical);const ht=toggleTaskInText(historical,0);
assert.equal(ht.text,historical.replace('- [ ] protected task','- [x] protected task'),'observation: exact first checkbox mutation, unrelated bytes retained');
const plain='## Current state\nA personal paragraph.\n- [ ] Plain personal task'; const pp=parseNote('Plain.md',plain);
assert.equal(pp.body,plain,'control: no-frontmatter remains entirely personal');assert.equal(toggleTaskInText(plain,0).label,'Plain personal task','control: plain task remains editable');
console.log(JSON.stringify({target:'1dc7f64',controlsPassed:7,historicalMutationAssertionPassed:true,canonicalProducerState:c.currentState,canonicalPersonalTask:tick.label,historicalCurrentState:h.currentState,historicalBodyContainsStipulatedProtectedTask:h.body.includes('protected task'),historicalFirstTickLabel:ht.label,policyOwnershipOfHistoricalParagraph:'not adopted; fixture intent alone cannot establish grammar',plainPersonalPreserved:true,crlfBytePreservation:true}));

const fs = require('node:fs');
const assert = require('node:assert/strict');
const root = process.argv[2];
const ts = require(process.argv[3]);
require.extensions['.ts'] = (m, p) => m._compile(ts.transpileModule(fs.readFileSync(p, 'utf8'), { compilerOptions: { module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2022 } }).outputText, p);
const store = () => { const m = new Map(); return { getItem: k => m.get(k) ?? null, setItem: (k,v) => m.set(k,String(v)), removeItem: k => m.delete(k), clear: () => m.clear() }; };
global.localStorage = store(); global.sessionStorage = store();
global.document = { addEventListener() {}, removeEventListener() {}, visibilityState: 'visible' };
const { FakeDrive } = require(root + '/src/data/drive/fake-drive.ts');
const { DriveVault } = require(root + '/src/data/vault/drive-vault.ts');
const { runFake } = require(root + '/src/data/fake/fake-runner.ts');
const { BowerError } = require(root + '/src/data/types.ts');
(async () => {
  const d = new FakeDrive();
  const r = await d.createFolder('root', 'Bower');
  for (const n of ['Inbox','Projects','Areas','Resources','Archive']) await d.createFolder(r.id,n);
  const sys = await d.createFolder(r.id,'.bower'); await d.createFolder(sys.id,'changes');
  const makeReader = () => new DriveVault({ drive: d, rootId: async () => r.id, runningPiles: () => new Set() });
  const uploader = makeReader();
  const upload = d.upload.bind(d);
  let refusals = 0;
  d.upload = async (parent, name, ...args) => {
    if (name === 'missing.pdf') { refusals++; throw new BowerError('offline', 'Synthetic upload is offline.'); }
    return upload(parent,name,...args);
  };
  const pile = await uploader.addPile({ text: 'Synthetic receipts', items: [
    { name: 'missing.pdf', kind: 'pdf', origin: 'Synthetic', body: new Blob(['missing']) },
    { name: 'arrived.pdf', kind: 'pdf', origin: 'Synthetic', body: new Blob(['arrived']) }
  ] });
  await uploader.settled();
  const before = (await uploader.piles()).find(p => p.id === pile.id);
  assert.equal(refusals, 1, 'control: one contract-conforming failed upload');
  assert.ok(before?.items.some(i => i.name === 'arrived.pdf'), 'control: successful sibling exists');
  assert.ok(before?.items.some(i => i.name === 'missing.pdf' && i.upload === -1), 'control: uploader records missing member');
  const second = makeReader();
  const seenBefore = (await second.piles()).find(p => p.id === pile.id);
  const result = await runFake(d, r.id, { scope: 'all', scale: 'instant', failOnce: false, onPhase: () => {} });
  await uploader.refresh(); await second.refresh();
  const afterUploader = (await uploader.piles()).find(p => p.id === pile.id);
  const afterSecond = (await second.piles()).find(p => p.id === pile.id);
  const files = await d.listAll(r.id);
  console.log(JSON.stringify({ target: '1dc7f64', scope: 'shared uploader and scripted fake consumer only', refusals,
    uploaderBefore: before.items.map(i => ({ name: i.name, ...(i.upload === undefined ? {} : { upload: i.upload }) })),
    secondBefore: seenBefore?.items.map(i => i.name) ?? null, phase: result.phase,
    filed: result.report?.filed.map(i => i.name), setAside: result.report?.setAside,
    uploaderAfter: afterUploader?.items.map(i => i.name) ?? null,
    secondAfter: afterSecond?.items.map(i => i.name) ?? null,
    arrivedCopies: files.filter(f => f.name === 'arrived.pdf').length,
    missingCopies: files.filter(f => f.name === 'missing.pdf').length }));
  assert.ok(afterSecond?.items.some(i => i.name === 'missing.pdf'), 'unready member must remain discoverable after another client tidy-up');
})().catch(e => { console.error(e); process.exitCode = 1; });
